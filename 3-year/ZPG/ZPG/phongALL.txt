#version 330 core

in vec3 worldPosition;
in vec3 worldNormal;
in vec2 TexCoord;  

out vec4 out_Color;

struct Material {
    float ra;          
    float rd;          
    float rs;          
    float shininess;
};

uniform Material material;

struct Light {
    int type;           
    vec3 position;     
    vec3 direction;     
    vec3 color;
    vec3 attenuation;
    float intensity;
    float cutOff;       
};

#define MAX_LIGHTS 10

uniform int numLights;
uniform Light lights[MAX_LIGHTS];
uniform vec3 objectColor;
uniform vec3 viewPosition;

uniform sampler2D  samplerTexture ; 

uniform vec3 ambientColor = vec3(1.0f, 1.0f, 1.0f);
uniform float ambientIntensity = 0.15f;

float calculateAttenuation(Light light, float distance) {
    return 1.0 / (
        light.attenuation.x +
        light.attenuation.y * distance +
        light.attenuation.z * distance * distance
    );
}


void main()
{
    vec3 N = normalize(worldNormal);
    vec3 V = normalize(viewPosition - worldPosition);
    vec3 result = vec3(0.0);

    vec3 ambient = material.ra * ambientColor * ambientIntensity;
    result = ambient * objectColor;

    float shininess = material.shininess;

    for (int i = 0; i < numLights; ++i)
    {
        vec3 L;
        float intensityFactor = 1.0;

        if (lights[i].type == 0) {
            // Point light
            L = normalize(lights[i].position - worldPosition);

            float distance = length(lights[i].position - worldPosition);
            float attenuation = calculateAttenuation(lights[i], distance);
            intensityFactor *= attenuation;
        } 
        else if (lights[i].type == 1) {
            // Directional light
            L = normalize(-lights[i].direction);
           
        } 
        else if (lights[i].type == 2) {
            // Spot light
           
           L = normalize(lights[i].position - worldPosition);

            float theta = dot(L, normalize(-lights[i].direction)); 
            float epsilon = 0.05; 
            float distance = length(lights[i].position - worldPosition);
           
            float att = 1.0;
            if (lights[i].attenuation.x > 0.0){
                att = calculateAttenuation(lights[i], distance);
            }
          
           intensityFactor *= smoothstep(lights[i].cutOff - epsilon, lights[i].cutOff, theta) * att;
           
        }

        float diff = max(dot(N, L), 0.0);
        vec3 diffuse = material.rd * diff * lights[i].color * lights[i].intensity * intensityFactor;


       vec3 specular = vec3(0.0);
        if(diff > 0){
            vec3 R = reflect(-L, N);
            float spec = pow(max(dot(R, V), 0.0), shininess);
            specular = material.rs * spec * lights[i].color * lights[i].intensity * intensityFactor;

        }

        result += (diffuse + specular);
    }
    vec3 texColor = texture( samplerTexture , TexCoord).rgb;
    out_Color = vec4(result * texColor * objectColor, 1.0);
  
}
